# 電力システムシミュレータ完全統合仕様書

## 1. 設計理念と目的
- **統一インターフェース**: すべてのコンポーネントが共通の`update()`メソッドを実装
- **高性能計算**: NumPyベースの状態管理とベクトル化演算
- **拡張可能アーキテクチャ**: 熱モデル、故障モデル、制御系を容易に追加
- **マルチスケールシミュレーション**: 電気・機械・熱の統合モデリング
- **実用性**: 実システムの動作条件・故障モードを再現

## 2. コアインターフェース仕様

### 2.1 SimulationPart基底クラス
```python
import numpy as np
from abc import ABC, abstractmethod

class SimulationPart(ABC):
    def __init__(self, name: str, params: dict = None):
        self.name = name
        self.params = params or {}
        self.state_vector = np.array([])  # 状態変数のベクトル
        self.state_metadata = {           # 状態メタデータ
            'index_map': {},               # 変数名→インデックスマッピング
            'units': {},                   # 変数単位
            'descriptions': {}             # 変数説明
        }
        self.last_input = None             # 前回入力データ
        self.last_output = None            # 前回出力データ
        self.last_time = 0.0               # 前回更新時間
        self.integrator = VectorIntegrator()  # 数値積分器
        self.differentiator = VectorDifferentiator()  # 数値微分器
    
    @abstractmethod
    def _process(self, input_data: np.ndarray, time: float, dt: float) -> np.ndarray:
        """コンポーネント固有の処理（サブクラス実装）"""
        pass
    
    def update(self, input_data: np.ndarray, time: float) -> np.ndarray:
        """統一更新インターフェース"""
        dt = time - self.last_time if self.last_time > 0 else 0.0
        self.last_time = time
        self.last_input = input_data
        self.last_output = self._process(input_data, time, dt)
        return self.last_output
    
    def get_state(self, var_name: str) -> float:
        """状態変数取得"""
        idx = self.state_metadata['index_map'].get(var_name, -1)
        return self.state_vector[idx] if 0 <= idx < len(self.state_vector) else 0.0
    
    def set_state(self, var_name: str, value: float):
        """状態変数設定"""
        idx = self.state_metadata['index_map'].get(var_name, -1)
        if 0 <= idx < len(self.state_vector):
            self.state_vector[idx] = value
    
    def reset(self):
        """状態リセット"""
        self.state_vector.fill(0)
        self.last_input = None
        self.last_output = None
        self.last_time = 0.0
        self.integrator.reset()
        self.differentiator.reset()
    
    def get_state_dict(self) -> dict:
        """状態を辞書形式で取得"""
        return {var: self.state_vector[idx] for var, idx in self.state_metadata['index_map'].items()}
```

## 3. 数値計算ライブラリ仕様

### 3.1 高効率ベクトル積分器
```python
class VectorIntegrator:
    METHODS = ['rk4', 'euler', 'bdf']
    
    def __init__(self, method: str = 'rk4', dt: float = 0.001):
        assert method in self.METHODS, f"無効な積分法: {method}"
        self.method = method
        self.dt = dt
        self.state = None
        self._buffers = {}
    
    def setup(self, initial_state: np.ndarray):
        """状態初期化とバッファ確保"""
        self.state = np.array(initial_state, dtype=np.float64, copy=True)
        
        # メソッド別バッファ確保
        if self.method == 'rk4':
            self._buffers = {
                'k1': np.empty_like(self.state),
                'k2': np.empty_like(self.state),
                'k3': np.empty_like(self.state),
                'k4': np.empty_like(self.state)
            }
    
    def update(self, derivatives: callable, time: float) -> np.ndarray:
        """状態更新"""
        if self.method == 'rk4':
            return self._rk4(derivatives, time)
        elif self.method == 'euler':
            return self._euler(derivatives, time)
        elif self.method == 'bdf':
            return self._bdf(derivatives, time)
    
    def _rk4(self, derivatives: callable, time: float) -> np.ndarray:
        """4次ルンゲクッタ法（メモリ最適化版）"""
        k1, k2, k3, k4 = self._buffers['k1'], self._buffers['k2'], self._buffers['k3'], self._buffers['k4']
        
        # 勾配計算（出力バッファを直接使用）
        derivatives(time, self.state, out=k1)
        derivatives(time + self.dt/2, self.state + k1*self.dt/2, out=k2)
        derivatives(time + self.dt/2, self.state + k2*self.dt/2, out=k3)
        derivatives(time + self.dt, self.state + k3*self.dt, out=k4)
        
        # 状態更新（インプレース演算）
        np.multiply(k1, 1/6, out=k1)
        np.multiply(k2, 1/3, out=k2)
        np.multiply(k3, 1/3, out=k3)
        np.multiply(k4, 1/6, out=k4)
        
        self.state += self.dt * (k1 + k2 + k3 + k4)
        return self.state
```

### 3.2 ベクトル微分器
```python
class VectorDifferentiator:
    METHODS = ['central', 'forward', 'backward']
    
    def __init__(self, method: str = 'central', dt: float = 0.001):
        self.method = method
        self.dt = dt
        self.prev_value = None
        self.prev_time = 0.0
        self._buffer = None
    
    def update(self, current_value: np.ndarray, current_time: float) -> np.ndarray:
        """微分計算"""
        if self.prev_value is None:
            self.prev_value = current_value.copy()
            self.prev_time = current_time
            if self._buffer is None:
                self._buffer = np.zeros_like(current_value)
            return self._buffer
        
        dt = current_time - self.prev_time
        if dt <= 0:
            return np.zeros_like(current_value)
        
        # 微分計算（メモリ再利用）
        if self.method == 'central' and hasattr(self, 'prev_prev_value'):
            # 中央差分（2次精度）
            np.subtract(current_value, self.prev_prev_value, out=self._buffer)
            np.divide(self._buffer, 2*dt, out=self._buffer)
        else:
            # 前方差分（1次精度）
            np.subtract(current_value, self.prev_value, out=self._buffer)
            np.divide(self._buffer, dt, out=self._buffer)
        
        # 状態更新
        if self.method == 'central':
            self.prev_prev_value = self.prev_value.copy()
        self.prev_value = current_value.copy()
        self.prev_time = current_time
        
        return self._buffer
```

## 4. コンポーネント接続システム

### 4.1 接続マネージャ
```python
import networkx as nx
import heapq

class ConnectionManager:
    def __init__(self):
        self.graph = nx.DiGraph()
        self.data_log = []
        self.connection_params = {}
    
    def add_connection(self, source: SimulationPart, target: SimulationPart, 
                      mapping: callable = None, delay: float = 0.0):
        """接続追加"""
        conn_id = f"{source.name}->{target.name}"
        self.graph.add_edge(source.name, target.name, id=conn_id, 
                          mapping=mapping, delay=delay)
        
        # 遅延バッファ初期化
        if delay > 0:
            self.connection_params[conn_id] = {
                'buffer': [],
                'timestamps': []
            }
    
    def propagate(self, time: float):
        """全接続のデータ伝播"""
        for source_name, target_name, data in self.graph.edges(data=True):
            source = self._get_component(source_name)
            target = self._get_component(target_name)
            conn_id = data['id']
            
            output = source.last_output
            if output is None:
                continue
            
            # マッピング適用
            if data['mapping']:
                input_data = data['mapping'](output)
            else:
                input_data = output
            
            # 遅延処理
            if data['delay'] > 0:
                self._add_to_buffer(conn_id, input_data, time)
                input_data = self._get_delayed_data(conn_id, time)
            
            # ターゲット更新
            if input_data is not None:
                target.update(input_data, time)
    
    def _add_to_buffer(self, conn_id: str, data: np.ndarray, time: float):
        """遅延バッファにデータ追加"""
        if conn_id not in self.connection_params:
            return
        
        buffer = self.connection_params[conn_id]['buffer']
        timestamps = self.connection_params[conn_id]['timestamps']
        
        buffer.append(data)
        timestamps.append(time)
        
        # バッファサイズ管理
        if len(buffer) > 1000:
            del buffer[0]
            del timestamps[0]
    
    def _get_delayed_data(self, conn_id: str, current_time: float) -> np.ndarray:
        """遅延データ取得"""
        if conn_id not in self.connection_params:
            return None
        
        buffer = self.connection_params[conn_id]['buffer']
        timestamps = self.connection_params[conn_id]['timestamps']
        delay = self.graph.edges[conn_id]['delay']
        
        target_time = current_time - delay
        if not timestamps or target_time < timestamps[0]:
            return None
        
        # 線形補間
        for i in range(1, len(timestamps)):
            if timestamps[i-1] <= target_time <= timestamps[i]:
                alpha = (target_time - timestamps[i-1]) / (timestamps[i] - timestamps[i-1])
                return buffer[i-1] + alpha * (buffer[i] - buffer[i-1])
        
        return buffer[-1]
```

## 5. 拡張シミュレーションマネージャ

### 5.1 AdvancedSimulationManager
```python
class AdvancedSimulationManager:
    EVENT_TYPES = ['COMPONENT', 'PARAM_CHANGE', 'FAULT']
    
    def __init__(self):
        self.components = {}
        self.connection_manager = ConnectionManager()
        self.time = 0.0
        self.history = []
        self.event_queue = []
        self.thermal_models = {}
        self.fault_models = {}
    
    def add_component(self, component: SimulationPart):
        """コンポーネント登録"""
        self.components[component.name] = component
        
        # 熱モデル/故障モデルを自動検出
        if hasattr(component, 'update_temperature'):
            self.thermal_models[component.name] = component
        if hasattr(component, 'apply_fault_effects'):
            self.fault_models[component.name] = component
    
    def add_event(self, time: float, event_type: str, target: str, action: callable):
        """イベント追加"""
        assert event_type in self.EVENT_TYPES, f"無効なイベントタイプ: {event_type}"
        heapq.heappush(self.event_queue, (time, {
            'type': event_type,
            'target': target,
            'action': action
        }))
    
    def run_simulation(self, t_total: float, dt: float, record_interval: float = 0.001):
        """シミュレーション実行"""
        num_steps = int(t_total / dt)
        record_every = max(1, int(record_interval / dt))
        
        for step in range(num_steps):
            self.time = step * dt
            
            # イベント処理
            self._process_events()
            
            # 入力コンポーネント更新（電源系）
            for comp in self.components.values():
                if isinstance(comp, PowerSource):
                    comp.update(None, self.time)
            
            # 接続伝播
            self.connection_manager.propagate(self.time)
            
            # 熱モデル更新
            for model in self.thermal_models.values():
                model.update_temperature(self.time)
            
            # 状態記録
            if step % record_every == 0:
                self._record_state()
    
    def _process_events(self):
        """キュー内のイベント処理"""
        while self.event_queue and self.event_queue[0][0] <= self.time:
            event_time, event = heapq.heappop(self.event_queue)
            target = self.components.get(event['target'])
            
            if target:
                if event['type'] == 'COMPONENT':
                    event['action'](target)
                elif event['type'] == 'PARAM_CHANGE':
                    target.params.update(event['action'](target.params))
                elif event['type'] == 'FAULT':
                    target.add_fault(**event['action'])
```

### 6.1 熱モデルインターフェース
```python
class ThermalModel:
    def __init__(self, thermal_params: dict):
        self.thermal_params = thermal_params
        self.temperature = thermal_params.get('ambient_temp', 298.15)
        self.heat_accumulated = 0.0
        self.cooling_rate = 0.0
        
    def calculate_power_loss(self, component: SimulationPart) -> float:
        """電力損失計算（コンポーネント固有）"""
        raise NotImplementedError("サブクラスで実装")
    
    def update_temperature(self, time: float):
        """温度更新"""
        # 熱平衡方程式: dT/dt = (P_loss - P_cool) / C_thermal
        dt = time - self.last_thermal_update
        power_loss = self.calculate_power_loss()
        cooling_power = self.cooling_rate * (self.temperature - self.ambient_temp)
        
        delta_T = (power_loss - cooling_power) / self.thermal_params['thermal_capacity']
        self.temperature += delta_T * dt
        self.last_thermal_update = time
        
        # 過熱保護
        if self.temperature > self.thermal_params['max_temp']:
            self.trigger_overheat_protection()
    
    def trigger_overheat_protection(self):
        """過熱保護作動"""
        # デフォルト実装: 出力を50%削減
        if hasattr(self, 'last_output'):
            self.last_output *= 0.5
```

```python
class InductionMotor(SimulationPart, ThermalModel):
    def __init__(self, name, params):
        SimulationPart.__init__(self, name, params.get('electrical', {}))
        ThermalModel.__init__(self, params.get('thermal', {}))
        
        # 状態ベクトルに温度を追加
        self.state_vector = np.zeros(len(self.STATE_VARS) + 1)
        self.state_metadata['index_map']['temperature'] = len(self.STATE_VARS)
        self.state_metadata['units']['temperature'] = 'K'
        
    def calculate_power_loss(self) -> float:
        """誘導電動機の損失計算"""
        # 銅損: I²R
        i_alpha = self.get_state('i_alpha')
        i_beta = self.get_state('i_beta')
        copper_loss = 1.5 * (i_alpha**2 * self.params['R1'] + 
                             i_beta**2 * self.params['R1'])
        
        # 鉄損: 周波数依存
        omega_e = self.get_state('omega') * self.derived_params['poles_pairs']
        iron_loss = self.thermal_params['iron_loss_coeff'] * omega_e**1.5
        
        return copper_loss + iron_loss
    
    def _process(self, input_data, time, dt):
        # 通常の電気・機械処理
        output = super()._process(input_data, time, dt)
        
        # 状態ベクトルに温度を設定
        self.set_state('temperature', self.temperature)
        
        # 出力に温度を追加
        return np.append(output, self.temperature)
```

## 7. 故障モデリングシステム

### 7.1 故障モード定義
```python
class FaultModes:
    SHORT_CIRCUIT = "short_circuit_phase"
    OPEN_CIRCUIT = "open_circuit_phase"
    OVERHEAT = "overheat"
    BEARING_DAMAGE = "bearing_damage"
    SENSOR_FAILURE = "sensor_failure"
    CONTROL_FAILURE = "control_failure"

class FaultModel:
    def __init__(self):
        self.active_faults = {}
        self.fault_probabilities = {
            FaultModes.SHORT_CIRCUIT: 0.001,
            FaultModes.OVERHEAT: 0.005
        }
    
    def add_fault(self, fault_type: str, severity: float = 1.0, 
                 start_time: float = 0, duration: float = float('inf')):
        """故障注入"""
        self.active_faults[fault_type] = {
            'severity': np.clip(severity, 0, 1),
            'start_time': start_time,
            'end_time': start_time + duration
        }
    
    def apply_fault_effects(self, nominal_value: np.ndarray, time: float) -> np.ndarray:
        """故障効果適用"""
        for fault, params in self.active_faults.items():
            if params['start_time'] <= time < params['end_time']:
                nominal_value = self._apply_fault(nominal_value, fault, params['severity'])
        return nominal_value
    
    def _apply_fault(self, value: np.ndarray, fault_type: str, severity: float) -> np.ndarray:
        """故障タイプ別処理"""
        if fault_type == FaultModes.SHORT_CIRCUIT:
            return value * (1 - 0.8 * severity)
        elif fault_type == FaultModes.OVERHEAT:
            return value * (1 + 0.2 * severity * np.random.randn())
        elif fault_type == FaultModes.SENSOR_FAILURE:
            return np.zeros_like(value)  # センサ出力喪失
        return value
```

```python
import matplotlib.pyplot as plt
from matplotlib.animation import FuncAnimation

class RealTimeVisualizer:
    def __init__(self, manager, update_interval=0.1):
        self.manager = manager
        self.fig, self.axes = plt.subplots(3, 1, figsize=(12, 10))
        self.lines = {}
        self.update_interval = update_interval
        
    def setup_plots(self, plot_config):
        """プロット設定"""
        for i, config in enumerate(plot_config):
            ax = self.axes[i]
            ax.set_title(config['title'])
            ax.set_xlabel('Time [s]')
            ax.set_ylabel(config['ylabel'])
            ax.grid(True)
            
            lines = []
            for var in config['variables']:
                line, = ax.plot([], [], label=var['label'])
                lines.append(line)
            
            self.lines[config['id']] = lines
            ax.legend()
        
        self.time_data = []
        self.value_data = {k: [] for k in self.lines.keys()}
    
    def update_plot(self, frame):
        """アニメーション更新関数"""
        current_time = self.manager.time
        
        for plot_id, lines in self.lines.items():
            comp_name, var_name = plot_id.split('|')
            comp = self.manager.components.get(comp_name)
            
            if not comp:
                continue
            
            value = comp.get_state(var_name)
            self.value_data[plot_id].append(value)
        
        self.time_data.append(current_time)
        
        # データ更新
        for plot_id, lines in self.lines.items():
            values = self.value_data[plot_id]
            for i, line in enumerate(lines):
                line.set_data(self.time_data, values)
        
        # 軸調整
        for ax in self.axes:
            ax.relim()
            ax.autoscale_view()
        
        return [line for lines in self.lines.values() for line in lines]
    
    def start_visualization(self):
        """可視化開始"""
        ani = FuncAnimation(
            self.fig, self.update_plot, interval=self.update_interval*1000,
            blit=True, cache_frame_data=False
        )
        plt.tight_layout()
        plt.show()
```

   ```python
   # 非効率
   ```

### 10.2 プロファイリングツール
```python
class Profiler:
    def __init__(self):
        self.perf_data = {
            'update_times': [],
            'memory_usage': []
        }
    
    def profile_update(self, component, time):
        start = time.perf_counter()
        mem_before = self.get_memory_usage()
        
        # 更新処理
        output = component.update(component.last_input, time)
        
        # 計測記録
        self.perf_data['update_times'].append(time.perf_counter() - start)
        self.perf_data['memory_usage'].append(self.get_memory_usage() - mem_before)
        
        return output
    
    def generate_report(self):
        """パフォーマンスレポート生成"""
        avg_time = np.mean(self.perf_data['update_times'])
        max_mem = np.max(self.perf_data['memory_usage'])
        return f"平均更新時間: {avg_time*1000:.2f}ms, 最大メモリ増加: {max_mem/1024:.2f}KB"
```
