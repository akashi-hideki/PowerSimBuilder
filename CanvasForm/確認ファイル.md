        {
            // 選択されたコンポーネントを追加

            {

            }
        }

        {
            // 選択された項目を削除
            if (_canvasManager.SelectedComponent != null)
            {
                _canvasManager.RemoveComponent(_canvasManager.SelectedComponent);
            }
            else if (_canvasManager.SelectedConnection != null)
            {
                _canvasManager.RemoveConnection(_canvasManager.SelectedConnection);
            }
        }

        {
            var model = _canvasManager.SaveModel();
            var json = JsonConvert.SerializeObject(model, Formatting.Indented);
            File.WriteAllText("design.json", json);
        }
        
        private void loadButton_Click(object sender, EventArgs e)
        {
            var json = File.ReadAllText("design.json");
            var model = JsonConvert.DeserializeObject<CanvasModel>(json);
            _canvasManager.LoadModel(model);
        }

        {
            _canvasManager.Undo();
        }

        {
            _canvasManager.Redo();
        }

        {
            // Pythonコードを生成
            string pythonCode = _codeGenerator.GenerateCode(_canvasManager.GetAllComponents());

            previewForm.Show();

            RunPythonSimulation(pythonCode);
        }

        private void RunPythonSimulation(string code)
        {
            try
            {
                using (Py.GIL())
                {
                    dynamic scope = Py.CreateScope();
                    scope.Exec(code);
                    scope.run_simulation();
                }
                MessageBox.Show("シミュレーションが正常に完了しました", "成功", MessageBoxButtons.OK, MessageBoxIcon.Information);
            }
            catch (Exception ex)
            {
                MessageBox.Show($"シミュレーションエラー: {ex.Message}", "エラー", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

using System;
using System.Collections.Generic;
using System.Drawing;
using System.Windows.Forms;

namespace PowerSimBuilder
{
    public partial class DesignCanvas : Panel
    {
        public event EventHandler SelectionChanged;
        public event EventHandler<ComponentEventArgs> ComponentAdded;
        public event EventHandler<ComponentEventArgs> ComponentRemoved;
        public event EventHandler<ComponentEventArgs> ComponentSelected;
        public event EventHandler<ConnectionEventArgs> ConnectionAdded;
        public event EventHandler<ConnectionEventArgs> ConnectionRemoved;
        public event EventHandler<ConnectionEventArgs> ConnectionSelected;
        
        private Connection _currentConnection;
        private Point _dragStartPosition;
        private ComponentControl _draggedComponent;
        private bool _isDragging;
        
        public List<ComponentControl> Components { get; } = new List<ComponentControl>();
        public List<Connection> Connections { get; } = new List<Connection>(); // 接続管理リスト追加
        public ComponentControl SelectedComponent { get; private set; }
        
        public DesignCanvas()
        {
            InitializeComponent();
            DoubleBuffered = true;
        }
        
        protected override void OnPaint(PaintEventArgs e)
        {
            base.OnPaint(e);
            
            // 既存の接続を描画
            foreach (var connection in Connections) // Connectionsリストを使用
            {
                DrawConnection(e.Graphics, connection);
            }
            
            // 作成中の接続を描画
            if (_currentConnection != null && _currentConnection.EndComponent == null)
            {
                DrawTemporaryConnection(e.Graphics);
            }
        }
        
        private void DrawConnection(Graphics g, Connection connection)
        {
            var startPos = connection.StartComponent.GetPortLocation(connection.StartPort);
            var endPos = connection.EndComponent.GetPortLocation(connection.EndPort);
            
            using (var pen = new Pen(Color.Blue, 2))
            {
                g.DrawLine(pen, startPos, endPos);
            }
        }
        
        private void DrawTemporaryConnection(Graphics g)
        {
            var startPos = _currentConnection.StartComponent.GetPortLocation(_currentConnection.StartPort);
            var endPos = PointToClient(Cursor.Position);
            
            using (var pen = new Pen(Color.LightBlue, 2) { DashStyle = System.Drawing.Drawing2D.DashStyle.Dash })
            {
                g.DrawLine(pen, startPos, endPos);
            }
        }
        
        // 既存メソッド（ディスクリプタ使用）
        public void AddComponent(ComponentDescriptor descriptor, Point location)
        {
            var control = new ComponentControl(descriptor) { Location = location };
            AddComponent(control); // 新しいメソッド呼び出し
        }
        
        // 新しいコンポーネント追加メソッド
        public void AddComponent(ComponentControl control)
        {
            if (!Controls.Contains(control))
            {
                control.SelectedChanged += (s, e) => 
                {
                    SelectedComponent = (ComponentControl)s;
                    SelectionChanged?.Invoke(this, EventArgs.Empty);
                    ComponentSelected?.Invoke(this, new ComponentEventArgs((ComponentControl)s));
                };
                
                control.PortClicked += ComponentControl_PortClicked;
                
                Controls.Add(control);
                Components.Add(control);
                ComponentAdded?.Invoke(this, new ComponentEventArgs(control));
                Invalidate();
            }
        }
        
        // コンポーネント削除メソッド
        public void RemoveComponent(ComponentControl component)
        {
            if (Controls.Contains(component))
            {
                // 関連接続を削除
                var connectionsToRemove = new List<Connection>();
                foreach (var conn in Connections)
                {
                    if (conn.StartComponent == component || conn.EndComponent == component)
                    {
                        connectionsToRemove.Add(conn);
                    }
                }
                
                foreach (var conn in connectionsToRemove)
                {
                    RemoveConnection(conn);
                }
                
                // コンポーネント削除
                Controls.Remove(component);
                Components.Remove(component);
                ComponentRemoved?.Invoke(this, new ComponentEventArgs(component));
                Invalidate();
            }
        }
        
        private void ComponentControl_PortClicked(object sender, PortEventArgs e)
        {
            if (_currentConnection == null)
            {
                _currentConnection = new Connection
                {
                    StartComponent = (ComponentControl)sender,
                    StartPort = e.PortName
                };
            }
            else
            {
                _currentConnection.EndComponent = (ComponentControl)sender;
                _currentConnection.EndPort = e.PortName;
                
                AddConnection(_currentConnection); // 接続追加メソッド使用
                _currentConnection = null;
            }
        }
        
        // 接続追加メソッド
        public void AddConnection(Connection connection)
        {
            if (!Connections.Contains(connection))
            {
                Connections.Add(connection);
                connection.StartComponent.Connections.Add(connection);
                connection.EndComponent.Connections.Add(connection);
                ConnectionAdded?.Invoke(this, new ConnectionEventArgs(connection));
                Invalidate();
            }
        }
        
        // 接続削除メソッド
        public void RemoveConnection(Connection connection)
        {
            if (Connections.Contains(connection))
            {
                Connections.Remove(connection);
                connection.StartComponent?.Connections.Remove(connection);
                connection.EndComponent?.Connections.Remove(connection);
                ConnectionRemoved?.Invoke(this, new ConnectionEventArgs(connection));
                Invalidate();
            }
        }
        
        protected override void OnMouseDown(MouseEventArgs e)
        {
            base.OnMouseDown(e);
            
            if (e.Button == MouseButtons.Left)
            {
                bool componentClicked = false;
                
                // コンポーネントのドラッグ処理
                foreach (var component in Components)
                {
                    if (component.Bounds.Contains(e.Location))
                    {
                        _isDragging = true;
                        _draggedComponent = component;
                        _dragStartPosition = e.Location;
                        component.BringToFront();
                        componentClicked = true;
                        break;
                    }
                }
                
                // 接続選択処理（コンポーネントがクリックされなかった場合）
                if (!componentClicked)
                {
                    foreach (var connection in Connections)
                    {
                        Point start = connection.StartComponent.GetPortLocation(connection.StartPort);
                        Point end = connection.EndComponent.GetPortLocation(connection.EndPort);
                        
                        if (IsPointNearLine(e.Location, start, end))
                        {
                            ConnectionSelected?.Invoke(this, new ConnectionEventArgs(connection));
                            return;
                        }
                    }
                    // 接続が選択されなかった場合
                    ConnectionSelected?.Invoke(this, new ConnectionEventArgs(null));
                }
            }
            else if (e.Button == MouseButtons.Right)
            {
                _currentConnection = null;
                Invalidate();
            }
        }
        
        protected override void OnMouseMove(MouseEventArgs e)
        {
            base.OnMouseMove(e);
            
            if (_isDragging && _draggedComponent != null)
            {
                int deltaX = e.X - _dragStartPosition.X;
                int deltaY = e.Y - _dragStartPosition.Y;
                
                _draggedComponent.Left += deltaX;
                _draggedComponent.Top += deltaY;
                
                _dragStartPosition = e.Location;
                Invalidate();
            }
        }
        
        protected override void OnMouseUp(MouseEventArgs e)
        {
            base.OnMouseUp(e);
            _isDragging = false;
            _draggedComponent = null;
        }
        
        // 点と線の距離判定
        private bool IsPointNearLine(Point point, Point lineStart, Point lineEnd)
        {
            double distance = DistanceFromPointToLine(point, lineStart, lineEnd);
            return distance < 5; // 5ピクセル以内をヒット判定
        }
        
        // 点と線分の距離計算
        private double DistanceFromPointToLine(Point point, Point lineStart, Point lineEnd)
        {
            double lineLengthSquared = Math.Pow(lineEnd.X - lineStart.X, 2) + 
                                      Math.Pow(lineEnd.Y - lineStart.Y, 2);
            
            if (lineLengthSquared == 0) 
                return Distance(point, lineStart);
            
            double t = Math.Max(0, Math.Min(1, 
                ((point.X - lineStart.X) * (lineEnd.X - lineStart.X) + 
                 (point.Y - lineStart.Y) * (lineEnd.Y - lineStart.Y)) / lineLengthSquared));
            
            Point projection = new Point(
                (int)(lineStart.X + t * (lineEnd.X - lineStart.X)),
                (int)(lineStart.Y + t * (lineEnd.Y - lineStart.Y)));
            
            return Distance(point, projection);
        }
        
        // 2点間の距離計算
        private double Distance(Point a, Point b)
        {
            return Math.Sqrt(Math.Pow(a.X - b.X, 2) + Math.Pow(a.Y - b.Y, 2));
        }
    }
    
    public class Connection
    {
        public ComponentControl StartComponent { get; set; }
        public string StartPort { get; set; }
        public ComponentControl EndComponent { get; set; }
        public string EndPort { get; set; }
    }

    // イベント引数クラス
    public class ComponentEventArgs : EventArgs
    {
        public ComponentControl Component { get; }
        public ComponentEventArgs(ComponentControl component) => Component = component;
    }

    public class ConnectionEventArgs : EventArgs
    {
        public Connection Connection { get; }
        public ConnectionEventArgs(Connection connection) => Connection = connection;
    }
}

using System;
using System.Collections.Generic;
using System.Drawing;
using System.Windows.Forms;

namespace PowerSimBuilder
{
    public partial class ComponentControl : UserControl
    {
        public event EventHandler SelectedChanged;
        public event EventHandler<PortEventArgs> PortClicked;
        
        private bool _selected;
        private readonly Dictionary<string, Rectangle> _ports = new Dictionary<string, Rectangle>();
        private Point _dragOffset;
        
        public ComponentDescriptor Descriptor { get; }
        public List<Connection> Connections { get; } = new List<Connection>();
        public Dictionary<string, object> Parameters { get; } = new Dictionary<string, object>();
        
        public string InstanceName { get; set; } = "comp";
        
        public bool IsSelected
        {
            get => _selected;
            set
            {
                if (_selected != value)
                {
                    _selected = value;
                    Invalidate();
                    SelectedChanged?.Invoke(this, EventArgs.Empty);
                }
            }
        }
        
        public ComponentControl(ComponentDescriptor descriptor)
        {
            Descriptor = descriptor;
            InitializeComponent();
            InitializeParameters();
            InitializePorts();
        }
        
        private void InitializeParameters()
        {
            foreach (var param in Descriptor.Parameters)
            {
                Parameters[param.Name] = param.DefaultValue;
            }
        }
        
        private void InitializePorts()
        {
            _ports.Add("Input", new Rectangle(0, Height / 2 - 5, 10, 10));
            _ports.Add("Output", new Rectangle(Width - 10, Height / 2 - 5, 10, 10));
        }
        
        public Point GetPortLocation(string portName)
        {
            if (_ports.TryGetValue(portName, out var rect))
            {
                return new Point(Left + rect.X + rect.Width / 2, Top + rect.Y + rect.Height / 2);
            }
            return Location;
        }
        
        protected override void OnPaint(PaintEventArgs e)
        {
            base.OnPaint(e);
            
            // 背景
            e.Graphics.FillRectangle(Brushes.LightGray, ClientRectangle);
            
            // テキスト
            e.Graphics.DrawString(Descriptor.Name, Font, Brushes.Black, 10, 10);
            
            // 選択状態
            if (IsSelected)
            {
                using (var pen = new Pen(Color.Red, 2))
                {
                    e.Graphics.DrawRectangle(pen, 0, 0, Width - 1, Height - 1);
                }
            }
            
            // ポート
            foreach (var port in _ports)
            {
                e.Graphics.FillEllipse(Brushes.Green, port.Value);
            }
        }
        
        protected override void OnMouseDown(MouseEventArgs e)
        {
            base.OnMouseDown(e);
            
            // ポートクリックをチェック
            foreach (var port in _ports)
            {
                if (port.Value.Contains(e.Location))
                {
                    PortClicked?.Invoke(this, new PortEventArgs(port.Key));
                    return;
                }
            }
            
            // コンポーネント選択
            IsSelected = true;
            _dragOffset = e.Location;
            BringToFront();
        }
        
        protected override void OnMouseMove(MouseEventArgs e)
        {
            base.OnMouseMove(e);
            
            if (e.Button == MouseButtons.Left && IsSelected)
            {
                Location = new Point(
                    Left + e.X - _dragOffset.X,
                    Top + e.Y - _dragOffset.Y
                );
            }
        }
    }
    
    public class PortEventArgs : EventArgs
    {
        public string PortName { get; }
        
        public PortEventArgs(string portName)
        {
            PortName = portName;
        }
    }
}

using System.Collections.Generic;
using System.Drawing;
using System.Windows.Forms;

namespace PowerSimBuilder
{
    public partial class ComponentPalette : FlowLayoutPanel
    {
        public List<ComponentDescriptor> AvailableComponents { get; } = new List<ComponentDescriptor>();
        
        public ComponentPalette()
        {
            InitializeComponent();
            InitializeAvailableComponents();
        }
        
        private void InitializeAvailableComponents()
        {
            // 利用可能なコンポーネントを初期化
            AvailableComponents.Add(new ComponentDescriptor
            {
                Name = "電圧形インバータ",
                Type = "VoltageSourceInverter",
                Icon = CreateIcon(Color.Red),
                Category = "電源",
                Parameters = new List<ParameterDescriptor>
                {
                    new ParameterDescriptor { Name = "V_dc", Type = "double", DefaultValue = 300 },
                    new ParameterDescriptor { Name = "f_rated", Type = "double", DefaultValue = 60 }
                }
            });
            
            AvailableComponents.Add(new ComponentDescriptor
            {
                Name = "誘導電動機",
                Type = "InductionMotor",
                Icon = CreateIcon(Color.Blue),
                Category = "負荷",
                Parameters = new List<ParameterDescriptor>
                {
                    new ParameterDescriptor { Name = "R1", Type = "double", DefaultValue = 0.15 },
                    new ParameterDescriptor { Name = "Lm", Type = "double", DefaultValue = 0.008 }
                }
            });
            
            // その他のコンポーネント...
            
            // パレットにボタンを追加
            foreach (var component in AvailableComponents)
            {
                var button = new Button
                {
                    Text = component.Name,
                    Image = component.Icon,
                    ImageAlign = ContentAlignment.MiddleLeft,
                    TextAlign = ContentAlignment.MiddleRight,
                    Tag = component,
                    Height = 40,
                    Width = 180,
                    Margin = new Padding(5)
                };
                
                button.MouseDown += ComponentButton_MouseDown;
                Controls.Add(button);
            }
        }
        
        private Bitmap CreateIcon(Color color)
        {
            var bmp = new Bitmap(32, 32);
            using (var g = Graphics.FromImage(bmp))
            {
                g.FillRectangle(new SolidBrush(color), 0, 0, 32, 32);
                g.DrawRectangle(Pens.Black, 0, 0, 31, 31);
            }
            return bmp;
        }
        
        private void ComponentButton_MouseDown(object sender, MouseEventArgs e)
        {
            var button = (Button)sender;
            var component = (ComponentDescriptor)button.Tag;
            
            var data = new DataObject();
            data.SetData("ComponentDescriptor", component);
            
            button.DoDragDrop(data, DragDropEffects.Copy);
        }
    }
    
    public class ComponentDescriptor
    {
        public string Name { get; set; }
        public string Type { get; set; }
        public Image Icon { get; set; }
        public string Category { get; set; }
        public List<ParameterDescriptor> Parameters { get; set; }
    }
    
    public class ParameterDescriptor
    {
        public string Name { get; set; }
        public string Type { get; set; }
        public object DefaultValue { get; set; }
    }
}
